cmake_minimum_required(VERSION 3.21)
project(FontLoaderTests LANGUAGES CXX)
enable_testing()
find_package(Freetype REQUIRED)
add_executable(FontLoaderTests FontLoaderTests.cpp ../../MGLEngine.Shared/Resources/FreeTypeLoader.cpp)
target_include_directories(FontLoaderTests PRIVATE ../..)
target_compile_features(FontLoaderTests PRIVATE cxx_std_20)
target_link_libraries(FontLoaderTests PRIVATE Freetype::Freetype)
set(FONT_TEST_FILE "" CACHE FILEPATH "Font file used for loader checks")
if(FONT_TEST_FILE)
    add_test(NAME FontLoaderTests COMMAND FontLoaderTests "${FONT_TEST_FILE}")
endif()
